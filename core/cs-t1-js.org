*50. كيف تكتب دالة تحول رقم عشري إلى ثنائي؟*

#+begin_src javascript
// الإجابة:
function decimalToBinary(decimal) {
    let binary = "";
    while (decimal > 0) {
        binary = (decimal % 2) + binary;
        decimal = Math.floor(decimal / 2);
    }
    return binary || "0";
}
#+end_src
